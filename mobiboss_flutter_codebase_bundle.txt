enum NetworkProvider { mtn, telecel, at, gmoney }
enum TransactionType { cashIn, cashOut, airtime, floatTransfer }
enum TransactionStatus { completed, pending, failed, offlineQueued }

class Till {
  final String id;
  final NetworkProvider network;
  final String tillName;
  final String tillNumber;
  final String simSlot;
  final double currentFloat;
  final double minFloatAlert;
  final String ussdPrefix;
  final String agentCode;

  Till({
    required this.id,
    required this.network,
    required this.tillName,
    required this.tillNumber,
    required this.simSlot,
    required this.currentFloat,
    required this.minFloatAlert,
    required this.ussdPrefix,
    required this.agentCode,
  });

  Till copyWith({double? currentFloat}) {
    return Till(
      id: id,
      network: network,
      tillName: tillName,
      tillNumber: tillNumber,
      simSlot: simSlot,
      currentFloat: currentFloat ?? this.currentFloat,
      minFloatAlert: minFloatAlert,
      ussdPrefix: ussdPrefix,
      agentCode: agentCode,
    );
  }
}

class TransactionRecord {
  final String id;
  final String tillId;
  final NetworkProvider network;
  final TransactionType type;
  final double amount;
  final double fee;
  final double commission;
  final String customerPhone;
  final String reference;
  final String ussdString;
  final DateTime timestamp;
  final TransactionStatus status;

  TransactionRecord({
    required this.id,
    required this.tillId,
    required this.network,
    required this.type,
    required this.amount,
    required this.fee,
    required this.commission,
    required this.customerPhone,
    required this.reference,
    required this.ussdString,
    required this.timestamp,
    required this.status,
  });
}

class SmsRecord {
  final String id;
  final String sender;
  final String rawText;
  final NetworkProvider network;
  final TransactionType type;
  final double amount;
  final double fee;
  final String reference;
  final double? postBalance;
  final DateTime timestamp;

  SmsRecord({
    required this.id,
    required this.sender,
    required this.rawText,
    required this.network,
    required this.type,
    required this.amount,
    required this.fee,
    required this.reference,
    this.postBalance,
    required this.timestamp,
  });
}

class ReconciliationResult {
  final double physicalCashCount;
  final double expectedPhysicalCash;
  final double cashVariance;
  final double totalCashIn;
  final double totalCashOut;
  final double totalCommissions;
  final String status; // BALANCED, SHORTAGE, OVERAGE
  final String aiDiagnosis;

  ReconciliationResult({
    required this.physicalCashCount,
    required this.expectedPhysicalCash,
    required this.cashVariance,
    required this.totalCashIn,
    required this.totalCashOut,
    required this.totalCommissions,
    required this.status,
    required this.aiDiagnosis,
  });
}

import 'package:url_launcher/url_launcher.dart';
import '../models/momo_models.dart';

class UssdService {
  /// Compiles a standard Ghana Telecom USSD String (MTN *171#, Telecel *110#, AT *110#)
  static String buildUssdString({
    required Till till,
    required TransactionType type,
    required String customerPhone,
    required double amount,
  }) {
    final amtFormatted = amount.toStringAsFixed(0);

    switch (till.network) {
      case NetworkProvider.mtn:
        if (type == TransactionType.cashIn) {
          // MTN Deposit format: *171*1*1*phone*phone*amount#
          return '*171*1*1*$customerPhone*$customerPhone*$amtFormatted#';
        } else if (type == TransactionType.cashOut) {
          // MTN Cash Out authorization: *171*1*2*phone*amount#
          return '*171*1*2*$customerPhone*$amtFormatted#';
        } else {
          return '*171*3*1*$customerPhone*$amtFormatted#';
        }

      case NetworkProvider.telecel:
        if (type == TransactionType.cashIn) {
          return '*110*1*1*$customerPhone*$amtFormatted#';
        } else if (type == TransactionType.cashOut) {
          return '*110*2*$customerPhone*$amtFormatted#';
        } else {
          return '*110*3*$customerPhone*$amtFormatted#';
        }

      case NetworkProvider.at:
        return '*110*1*$customerPhone*$amtFormatted#';

      case NetworkProvider.gmoney:
        return '*422*1*$customerPhone*$amtFormatted#';
    }
  }

  /// Directly launches the device phone dialer with pre-filled USSD string
  static Future<bool> dialUssd(String ussdCode) async {
    // URL-encode '#' symbol as '%23' so Android/iOS dialer does not strip it
    final encodedCode = ussdCode.replaceAll('#', '%23');
    final Uri uri = Uri.parse('tel:$encodedCode');

    if (await canLaunchUrl(uri)) {
      return await launchUrl(uri);
    } else {
      return false;
    }
  }
}

import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import '../providers/momo_provider.dart';

import '../services/ussd_service.dart';

class TransactScreen extends StatefulWidget {
  const TransactScreen({super.key});

  @override
  State<TransactScreen> createState() => _TransactScreenState();
}

class _TransactScreenState extends State<TransactScreen> {
  TransactionType _selectedType = TransactionType.cashIn;
  final TextEditingController _phoneController = TextEditingController();
  final TextEditingController _amountController = TextEditingController();

  final List<double> _quickAmounts = [50, 100, 200, 500, 1000];

  @override
  void dispose() {
    _phoneController.dispose();
    _amountController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final provider = context.watch<MoMoProvider>();
    final activeTill = provider.selectedTill;
    final amount = double.tryParse(_amountController.text) ?? 0.0;

    final ussdString = UssdService.buildUssdString(
      till: activeTill,
      type: _selectedType,
      customerPhone: _phoneController.text.isEmpty ? '024XXXXXXX' : _phoneController.text,
      amount: amount,
    );

    return SingleChildScrollView(
      padding: const EdgeInsets.all(20),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          // 1. Till Selector Card
          Card(
            child: Padding(
              padding: const EdgeInsets.all(16),
              child: Row(
                children: [
                  Container(
                    width: 44,
                    height: 44,
                    decoration: BoxDecoration(
                      color: activeTill.network == NetworkProvider.mtn
                          ? const Color(0xFFFFF4CC)
                          : const Color(0xFFFFEBEE),
                      borderRadius: BorderRadius.circular(12),
                    ),
                    child: Icon(
                      Icons.sim_card_rounded,
                      color: activeTill.network == NetworkProvider.mtn
                          ? const Color(0xFFF9AB00)
                          : Colors.red,
                    ),
                  ),
                  const SizedBox(width: 16),
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Text(
                          activeTill.tillName,
                          style: const TextStyle(fontWeight: FontWeight.w700, fontSize: 14),
                        ),
                        Text(
                          'Float: GHS ${activeTill.currentFloat.toStringAsFixed(2)}',
                          style: const TextStyle(fontSize: 13, color: Color(0xFF1E8E3E), fontWeight: FontWeight.w600),
                        ),
                      ],
                    ),
                  ),
                  DropdownButton<String>(
                    value: activeTill.id,
                    underline: const SizedBox(),
                    items: provider.tills.map((t) {
                      return DropdownMenuItem(
                        value: t.id,
                        child: Text(t.network.name.toUpperCase()),
                      );
                    }).toList(),
                    onChanged: (val) {
                      if (val != null) provider.selectTill(val);
                    },
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 16),

          // 2. Transaction Type Segmented Pills
          SegmentedButton<TransactionType>(
            segments: const [
              ButtonSegment(
                value: TransactionType.cashIn,
                label: Text('Cash In (Deposit)'),
                icon: Icon(Icons.arrow_downward_rounded),
              ),
              ButtonSegment(
                value: TransactionType.cashOut,
                label: Text('Cash Out (Withdraw)'),
                icon: Icon(Icons.arrow_upward_rounded),
              ),
            ],
            selected: {_selectedType},
            onSelectionChanged: (set) => setState(() => _selectedType = set.first),
          ),
          const SizedBox(height: 16),

          // 3. Customer Phone Input
          TextField(
            controller: _phoneController,
            keyboardType: TextInputType.phone,
            decoration: InputDecoration(
              labelText: 'Customer Mobile Number',
              hintText: '0244123456',
              prefixIcon: const Icon(Icons.phone_android_rounded),
              border: OutlineInputBorder(borderRadius: BorderRadius.circular(12)),
              filled: true,
              fillColor: Colors.white,
            ),
          ),
          const SizedBox(height: 16),

          // 4. Amount Input
          TextField(
            controller: _amountController,
            keyboardType: const TextInputType.numberWithOptions(decimal: true),
            style: const TextStyle(fontSize: 24, fontWeight: FontWeight.bold),
            decoration: InputDecoration(
              labelText: 'Amount (GHS)',
              prefixText: 'GHS ',
              border: OutlineInputBorder(borderRadius: BorderRadius.circular(12)),
              filled: true,
              fillColor: Colors.white,
            ),
          ),
          const SizedBox(height: 12),

          // Quick Amount Chips
          Wrap(
            spacing: 8,
            children: _quickAmounts.map((amt) {
              return ActionChip(
                label: Text('+${amt.toInt()}'),
                onPressed: () {
                  setState(() {
                    _amountController.text = amt.toStringAsFixed(0);
                  });
                },
              );
            }).toList(),
          ),
          const SizedBox(height: 24),

          // 5. Live USSD Preview & Direct Dial Button
          Card(
            color: const Color(0xFFF1F3F4),
            child: Padding(
              padding: const EdgeInsets.all(16),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const Text(
                    'Compiled USSD String:',
                    style: TextStyle(fontSize: 12, color: Color(0xFF5F6368)),
                  ),
                  const SizedBox(height: 4),
                  SelectableText(
                    ussdString,
                    style: const TextStyle(
                      fontFamily: 'monospace',
                      fontWeight: FontWeight.bold,
                      fontSize: 16,
                      color: Color(0xFF1A73E8),
                    ),
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 20),

          // 6. Action Button: 1-Tap Dial & Record
          FilledButton.icon(
            style: FilledButton.styleFrom(
              padding: const EdgeInsets.symmetric(vertical: 16),
              backgroundColor: const Color(0xFF1A73E8),
              shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
            ),
            icon: const Icon(Icons.phone_in_talk_rounded),
            label: const Text('Dial USSD & Record Transaction', style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold)),
            onPressed: () async {
              if (amount <= 0 || _phoneController.text.length < 9) {
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(content: Text('Please enter a valid phone number and amount')),
                );
                return;
              }

              // Dial USSD on hardware
              await UssdService.dialUssd(ussdString);

              // Record in local database
              provider.recordTransaction(
                type: _selectedType,
                amount: amount,
                customerPhone: _phoneController.text,
                ussdString: ussdString,
              );

              ScaffoldMessenger.of(context).showSnackBar(
                const SnackBar(
                  content: Text('Transaction recorded! Float updated.'),
                  backgroundColor: Color(0xFF1E8E3E),
                ),
              );

              _amountController.clear();
              _phoneController.clear();
            },
          ),
        ],
      ),
    );
  }
}
